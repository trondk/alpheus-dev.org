{{- /* URL of a file in the primary repository, or the repository itself for "" (WWW-19). Forgejo and GitHub lay out file URLs differently. */ -}} {{- $repo := site.Params.repo -}} {{- $url := $repo -}} {{- with . -}} {{- if strings.Contains $repo "github.com" -}} {{- $url = printf "%s/blob/%s/%s" $repo site.Params.repoBranch . -}} {{- else -}} {{- $url = printf "%s/src/branch/%s/%s" $repo site.Params.repoBranch . -}} {{- end -}} {{- end -}} {{- return $url -}}
